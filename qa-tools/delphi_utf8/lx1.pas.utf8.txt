    for ii:=1 to count do
      if recebuf[ii]=$a5 then
        begin
          form1.Timer3.enabled:=false;

            4:
              begin

              end;
            1:

                form1.hide;
                form2.show;
                form2.button1.setfocus;

                data_come:=false;
                form1.Timer4.interval:=1000;
                form1.Timer4.enabled:=true;
              end;
            2:
              begin
                form2.hide;
                form1.show;
                tishi.simpletext:='';
              end;
            3:
              begin
                for i:=0 to 7 do
                  for j:=0 to 3 do
                    form5.stringgrid2.Cells[i,j]:='';
                form5.stringgrid2.col:=0;
                form5.stringgrid2.row:=0;
                form5.edit2.text:='0';
                grid2_lie:=0;
                grid2_hang:=0;
                v2i_h:=0;
                v2i_l:=0;
                gdcs:=0;
                dots:=1;
                form5.tishi.simpletext:='正在供电、测量...';
                ke_cunpan:=false;
                data_come:=false;
                form1.Timer4.interval:=6000;
                form1.Timer4.enabled:=true;
              end;

          for ij:=1 to count-1 do
            if (recebuf[ij]=$a0) and (recebuf[ij+1]=$a0) then
              begin
                msg_ii:=1;
                buf_ii:=ij+2;
                have_head:=true;
              end;
          if have_head and (buf_ii<=count) then
            for ij:=buf_ii to count do
              begin
                msg[msg_ii]:=recebuf[ij];
                if (msg[msg_ii-1]=$af) and (msg[msg_ii]=$af) then
                  begin
                    have_head:=false;
                    check_byte:=0;
                    for j:=1 to msg_ii-2 do
                      check_byte:=check_byte xor msg[j];
                    if (check_byte=$0) or (check_byte=$80) then
                      begin
                        if msg[1]<>1 then
                          begin
                            send[1]:=$a5;
                            for j:=1 to 16 do
                              Comm1.write(send,1);
                          end;
                        case msg[1] of
                        4:
                          begin
                            form1.hide;
                            form6.show;
                            vab:=((msg[2] and $3)*256+msg[3])/1023*50;
                            form6.trackbar1.position:=trunc(vab/60*120);
                            vab:=trunc(vab*100)/100;
                            form6.edit1.text:=floattostr(vab);
                            data_come:=true;
                            if ((msg[2] and $3)*256+msg[3])/1023*50>=strtofloat(form6.edit2.text) then
                              begin
                                form6.tishi.simpletext:='充电完毕 !';
                                form6.button2.setfocus;
                              end;
                          end;

                        1:
                          begin
                            form1.hide;
                            form2.show;
                            form2.button1.setfocus;
                            form2.tishi.simpletext:='正在自检...';
                            vab:=((msg[2] and $3)*256+msg[3])/1023*50;
                            form2.trackbar1.position:=trunc(vab/60*120);
                            vab:=trunc(vab*100)/100;
                            form2.edit1.text:=floattostr(vab);
                            dianchi:=((msg[4] and $3)*256+msg[5])/1023*50.45; //*105.5;
                            dianchi:=trunc(dianchi*100)/100;
                            form2.edit2.text:=floattostr(dianchi);
                            data_come:=true;
                          end;

                            data_come:=true;

procedure TForm3.Button2Click(Sender: TObject);
var
  F: TextFile;
  k,i,j,sum:integer;
begin
  // prg_no:=2;  // strtoint(combobox1.text)  2003-6-6
  prg_no:=strtoint(combobox1.text); // 2005-11-19, 2006-9-3, 2006-11-19
  case prg_no of
  1: begin
       points:=22;
     end;
  2: begin
       points:=28;
     end;
  3: begin
       points:=34;
     end;
  4: begin
       points:=40;
     end;
  end;
      k:=1;
      shjian[1]:=1*0.08;    //prog_no=1,2,3,4
    {
        if points>=22 then    //prog_no=4, no use
        begin
          k:=2;
          shjian[1]:=2*0.08;
        end;
     }
      for i:=2 to points do
        begin
          sum:=0;
          for j:=1 to i do
            begin
              k:=k+1;
              sum:=sum+k;
            end;
          shjian[i]:=sum/i*0.08;
        end;
  AssignFile(F,wjljv+'zygs'+'.txt');  //  +combobox1.text  2003-6-6
  Rewrite(F); //single/dauble wave,dao shu,bian kuang,zhongxin/chongdit
  if jsfs.text='重叠' then writeln(F,2,' ',points,' ',strtoint(gongquan.text),' ',0)
    else writeln(F,2,' ',points,' ',strtoint(gongquan.text),' ',1);
  if jsfs.text='中心' then writeln(F,strtoint(cequan.text));//jieshou mianji
  writeln(F,4,' ',0.02);//gongdian maikuan,guanduan shijian
  for i:=1 to points do
    writeln(F,shjian[i]);
  CloseFile(F);
end;

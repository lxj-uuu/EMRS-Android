procedure TForm5.replace_ave();
var
  i,j,jj,k,i_bad:integer;
  sum,ave,temp,dlt:real;
  g1,g2,g3,g4,g5,g6:real;
  b78,b67,b56,b45,b34,b23,b12:real;
  v6p,v5p,v4p,v3p,v2p,v1p:real;
  rplc:array[0..32] of real;
label  loop_rplc;
label  rplc_j_ok;
label  rplc_ok;
begin
  if exprmt_times=1 then
    begin
      for j:=1 to points_end1 do
        v2[j]:=v2value[1,j];
      goto rplc_ok;
    end;
  for j:=1 to points_end1 do
  begin
    sum:=0;
    for i:=1 to exprmt_times do
      sum:=sum+v2value[i,j];
    ave:=sum/exprmt_times;
    sum:=0;
    for i:=1 to exprmt_times do
      sum:=sum+(v2value[i,j]-ave)*(v2value[i,j]-ave);
    dlt:=sqrt(sum/(exprmt_times-1));
    k:=0;
    for i:=1 to exprmt_times do
      if abs(v2value[i,j]-ave)<=3*dlt then
        begin
          k:=k+1;
          rplc[k]:=v2value[i,j];
        end;
    if k<2 then  goto rplc_j_ok;
loop_rplc:
    sum:=0;
    for i:=1 to k do
      sum:=sum+rplc[i];
    ave:=sum/k;
    dlt:=0;
    for i:=1 to k do
      if abs(rplc[i]-ave)>dlt then
        begin
          dlt:=abs(rplc[i]-ave);
          i_bad:=i;
        end;
 //   if dlt/abs(ave)>0.02 then
    if dlt>abs(ave)*0.02 then
      begin
        rplc[i_bad]:=ave;
        goto loop_rplc;
      end;
rplc_j_ok:
    v2[j]:=ave;
  end;
rplc_ok:

//  降序 ---- 不要了
//  for j:=1 to points_end-1 do
//    for jj:=j+1 to points_end do
//      if v2[j]<v2[jj] then
//        begin
//          temp:=v2[j];
//          v2[j]:=v2[jj];
//          v2[jj]:=temp;
//        end;

{ 修尖 ---- 不要了
  g6:=1.02;
  g5:=1.06;
  g4:=1.12;
  g3:=1.3;
  g2:=1.5;
  g1:=2;

  b78:=v2[7]/v2[8];
  b67:=b78*g6;
  b56:=b67*g5;
  b45:=b56*g4;
  b34:=b45*g3;
  b23:=b34*g2;
  b12:=b23*g1;

  v6p:=b67*v2[7];
  v5p:=b56*v6p;
  v4p:=b45*v5p;
  v3p:=b34*v4p;
  v2p:=b23*v3p;
  v1p:=b12*v2p;

  v2[6]:=v6p;
  v2[5]:=v5p;
  v2[4]:=v4p;
  v2[3]:=v3p;
  v2[2]:=v2p;
  v2[1]:=v1p;
}
end;

  i,j,k:integer;
  sum:real;
begin
  k:=1;

    begin
      sum:=0;
      for j:=1 to i do
        begin
          k:=k+1;
          sum:=sum+v2[k];
        end;
      v2i[i]:=sum/i;
    end;
end;
